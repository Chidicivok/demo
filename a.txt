        -->
        <java.version>21</java.version>

    </properties>


    <dependencies>


        <!-- ===================================================== -->
        <!-- SPRING WEB                                             -->
        <!-- ===================================================== -->

        <!--
            Gives us:

            @Controller
            @RestController
            @RequestMapping
            @GetMapping
            @PostMapping
            @PutMapping
            @PatchMapping

            Your static HTML files inside:

            src/main/resources/static

            are also served by Spring Boot.
        -->
        <dependency>

            <groupId>org.springframework.boot</groupId>

            <artifactId>spring-boot-starter-web</artifactId>

        </dependency>



        <!-- ===================================================== -->
        <!-- SPRING DATA JPA                                       -->
        <!-- ===================================================== -->

        <!--
            Gives us:

            @Entity
            @Table
            @Id
            @GeneratedValue
            @Column
            @ManyToOne
            @JoinColumn

            JpaRepository

            This is what allows UserAccountRepository
            and OTPRepository to communicate with H2.
        -->
        <dependency>

            <groupId>org.springframework.boot</groupId>

            <artifactId>spring-boot-starter-data-jpa</artifactId>

        </dependency>

        <!-- H2 DATABASE                                           -->
        <!-- ===================================================== -->

        <!--
            H2 is the database we are using for this project.

            scope=runtime means:
            Java code does not need to directly compile against
            H2 classes, but H2 must be available when the
            application runs.
        -->
        <dependency>

            <groupId>com.h2database</groupId>

            <artifactId>h2</artifactId>

            <scope>runtime</scope>

        </dependency>



        <!-- ===================================================== -->
        <!-- VALIDATION                                            -->
        <!-- ===================================================== -->

        <!--
            Gives us validation annotations such as:

            @NotBlank
            @NotNull
            @Size
            @Pattern

            These are used heavily inside your request DTOs.
        -->
        <dependency>

            <groupId>org.springframework.boot</groupId>

            <artifactId>spring-boot-starter-validation</artifactId>

        </dependency>



        <!-- ===================================================== -->
        <!-- SPRING SECURITY CRYPTO                                -->
        <!-- ===================================================== -->

        <!--
            IMPORTANT:

            We are NOT using Spring Security to control login
            and authorization in this version of the project.

            We are only borrowing its secure password utilities:

            PasswordEncoder
            BCryptPasswordEncoder

            This allows us to safely hash passwords rather than
            storing plain-text passwords in H2.
        -->
        <dependency>

            <groupId>org.springframework.security</groupId>

            <artifactId>spring-security-crypto</artifactId>

        </dependency>



        <!-- ===================================================== -->
        <!-- LOMBOK                                                -->
        <!-- ===================================================== -->

        <!-- ===================================================== -->
